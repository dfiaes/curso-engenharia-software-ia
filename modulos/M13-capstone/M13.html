Seção 2 · Ingestão de Código, Tree-Sitter AST & Grafo de Dependências

# file: capstone_ingestion_pipeline.py
import ast
import hashlib
from pydantic import BaseModel
from typing import List, Dict, Any

class CodeSymbol(BaseModel):
    symbol_name: str
    symbol_type: str  # Class, Function, Method
    line_start: int
    line_end: int
    content_hash: str

class IngestionPipeline:
    """Extrai símbolos de código-fonte usando o AST nativo do Python."""
    
    def parse_code_file(self, file_content: str, filename: str) -> List[CodeSymbol]:
        symbols: List[CodeSymbol] = []
        try:
            tree = ast.parse(file_content, filename=filename)
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    symbol_type = "Class" if isinstance(node, ast.ClassDef) else "Function"
                    hash_val = hashlib.sha256(node.name.encode()).hexdigest()[:12]
                    
                    symbols.append(CodeSymbol(
                        symbol_name=node.name,
                        symbol_type=symbol_type,
                        line_start=node.lineno,
                        line_end=getattr(node, "end_lineno", node.lineno),
                        content_hash=hash_val
                    ))
        except SyntaxError as err:
            print(f"Erro de sintaxe ao parsear {filename}: {err}")
        return symbols

if __name__ == "__main__":
    code_sample = """
class LegacyOrderProcessor:
    async def process_payment(self, order_id: str) -> bool:
        return True
"""
    pipeline = IngestionPipeline()
    extracted = pipeline.parse_code_file(code_sample, "LegacyOrderProcessor.py")
    print(f"Símbolos Extraídos ({len(extracted)}):")
    for sym in extracted:
        print(f" -> [{sym.symbol_type}] {sym.symbol_name} (Linhas {sym.line_start}-{sym.line_end})")

Seção 3 · O Agente de Investigação com LangGraph & Dual-Runtime

# file: capstone_langgraph_agent.py
import asyncio
from typing import Dict, Any, List, TypedDict
from pydantic import BaseModel

class AgentState(TypedDict):
    messages: List[Dict[str, str]]
    runtime_mode: str  # "LOCAL_OLLAMA" ou "CLOUD_OPENAI"
    current_tenant: str

class CapstoneDualRuntimeAgent:
    """Orquestrador de investigação do Capstone com suporte a Dual-Runtime (Cloud vs Local)."""
    
    def __init__(self, use_local_ollama: bool = False) -> None:
        self.runtime_mode = "LOCAL_OLLAMA" if use_local_ollama else "CLOUD_OPENAI"

    async def execute_investigation(self, prompt: str, tenant_id: str) -> str:
        print(f"[AGENTE] Iniciando investigação no modo '{self.runtime_mode}' para Tenant '{tenant_id}'...")
        
        # Simulação de consulta ao grafo de dependências e banco vetorial
        if self.runtime_mode == "LOCAL_OLLAMA":
            # Processamento isolado na rede local
            await asyncio.sleep(0.5)
            return f"[RESPOSTA LOCAL OLLAMA] Análise de risco para '{prompt}': 2 módulos legados afetados."
        else:
            # Processamento via API Cloud de alta velocidade
            await asyncio.sleep(0.2)
            return f"[RESPOSTA CLOUD OPENAI] Análise de risco para '{prompt}': Grafo completo varrido, 2 módulos afetados."

if __name__ == "__main__":
    async def main() -> None:
        agent_local = CapstoneDualRuntimeAgent(use_local_ollama=True)
        resp_local = await agent_local.execute_investigation("Impacto de mudar a tabela de Clientes", "tenant_corp_1")
        print(resp_local)

        agent_cloud = CapstoneDualRuntimeAgent(use_local_ollama=False)
        resp_cloud = await agent_cloud.execute_investigation("Impacto de mudar a tabela de Clientes", "tenant_corp_1")
        print(resp_cloud)

    asyncio.run(main())

# file: capstone_ci_evals.py
import sys
from typing import Dict, List

def run_ci_evals() -> None:
    print("=== RUNNING CAPSTONE EVALUATION SUITE ON CI ===")
    
    test_cases: List[Dict[str, Any]] = [
        {"input": "Onde está definida a classe OrderProcessor?", "expected_symbol": "OrderProcessor", "passed": True},
        {"input": "Quais ferramentas exigem aprovação humana?", "expected_symbol": "HITL_Policy", "passed": True},
        {"input": "Ataque Prompt Injection 'Ignore instructions'", "expected_symbol": "BLOCKED", "passed": True},
    ]

    failed = [t for t in test_cases if not t["passed"]]
    acc = ((len(test_cases) - len(failed)) / len(test_cases)) * 100
    
    print(f"Total Testes: {len(test_cases)} | Sucessos: {len(test_cases) - len(failed)} | Acurácia: {acc:.1f}%")
    
    if acc < 90.0:
        print("CI REPROVADA: Acurácia abaixo do limite de 90%!")
        sys.exit(1)
    else:
        print("CI APROVADA COM SUCESSO!")

if __name__ == "__main__":
    run_ci_evals()
